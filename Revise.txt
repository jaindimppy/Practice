public class Student
{
    public int Id { get; set; }
    public string Name { get; set; } = string.Empty;
    public int Age { get; set; }

    // Navigation Property: One student has many subject marks
    public ICollection<SubjectMark> Marks { get; set; } = new List<SubjectMark>();
}

public class SubjectMark
{
    public int Id { get; set; }
    public string SubjectName { get; set; } = string.Empty; // e.g., "Hindi", "English", "Maths"
    public int Score { get; set; }

    // Foreign Key
    public int StudentId { get; set; }
    public Student Student { get; set; } = null!;
}

// 1. Instantiate (no manual IDs needed)
var student = new Student
{
    Name = "Jain",
    Age = 25,
    Marks = new List<SubjectMark>
    {
        new SubjectMark { SubjectName = "Hindi", Score = 60 },
        new SubjectMark { SubjectName = "English", Score = 90 },
        new SubjectMark { SubjectName = "Maths", Score = 100 }
    }
};


Find departments where average salary is greater than 60,000.
SELECT deptId, AVG(salary) AS avgSalary  FROM employee GROUP BY deptId 
HAVING AVG(salary) > 60000;

Get employees along with their manager names (same table).
SELECT e.name AS employeeName, m.name AS managerName  FROM employee e  LEFT JOIN employee m  ON e.managerId = m.id;


Find total salary paid per department.
SELECT deptId, SUM(salary) AS totalSalary  FROM employee GROUP BY deptId;

Find employees with the same salary.
SELECT 
    e.Id AS Employee1_Id,
    e.Name AS Employee1_Name,
    e1.Id AS Employee2_Id,
    e1.Name AS Employee2_Name,
    e.Salary
FROM Employee e
INNER JOIN Employee e1 
    ON e.Salary = e1.Salary 
   AND e.Id < e1.Id;
